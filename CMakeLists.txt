# sing-vst. Decisions: plan/DECISIONS.md (D-001, D-002, D-024). Pins: plan/ENVIRONMENT.md.
cmake_minimum_required(VERSION 3.22)
project(SingVst VERSION 0.1.0 LANGUAGES C CXX)   # JUCE needs C (C-012, PK-37)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(SINGVST_BUILD_PLUGIN "Build the JUCE VST3/Standalone plugin (needs JUCE)" ON)
option(SINGVST_BUILD_TESTS "Build the Catch2 test executables" ON)
set(SINGVST_SANITIZE "" CACHE STRING "Sanitizers for the core and tests: '', 'address;undefined' or 'thread'")
if(MSVC)
    add_compile_options(/utf-8 /permissive- /W4 /Zc:__cplusplus)
else()
    add_compile_options(-Wall -Wextra)
endif()
if(SINGVST_SANITIZE)
    list(JOIN SINGVST_SANITIZE "," _san)
    add_compile_options(-fsanitize=${_san} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    add_link_options(-fsanitize=${_san})
endif()
include(cmake/Dependencies.cmake)
add_subdirectory(core)
add_subdirectory(cli)
if(SINGVST_BUILD_PLUGIN)
    add_subdirectory(plugin)
endif()
if(SINGVST_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests/cpp)
endif()
